cmake_minimum_required(VERSION 3.22)
project(ZoneEditMobile NONE)

# No implicit host compiler, Gradle, javac, Kotlin, d8 or Smali stage.
foreach(required ANDROID_SDK_ROOT IDRIC_EXECUTABLE IDRIC_SUPPORT_DIR IDRIC_SCHEME DEX_BACKEND_DIR AICI_DIR)
  if(NOT DEFINED ${required} OR NOT EXISTS "${${required}}")
    message(FATAL_ERROR "${required} must identify an existing build-host dependency")
  endif()
endforeach()
set(ndk "${ANDROID_SDK_ROOT}/ndk/27.2.12479018/toolchains/llvm/prebuilt/linux-x86_64/bin")
set(tools "${ANDROID_SDK_ROOT}/build-tools/35.0.0")
set(stage "${CMAKE_BINARY_DIR}/apk")
set(generator_dir "${CMAKE_BINARY_DIR}/generator")
set(paths "${CMAKE_BINARY_DIR}/backend/ttc:${IDRIC_SUPPORT_DIR}/libs/prelude/build/ttc:${IDRIC_SUPPORT_DIR}/libs/base/build/ttc")
set(compiler_env "IDRIS2_PATH=${paths}" "IDRIS2_DATA=${IDRIC_SUPPORT_DIR}/support"
  "IDRIS2_LIBS=${IDRIC_SUPPORT_DIR}/support/chez" "CHEZ=${IDRIC_SCHEME}")
file(GLOB backend_sources CONFIGURE_DEPENDS "${DEX_BACKEND_DIR}/src/Backend/DEX/*.idr")
set(activity "${CMAKE_SOURCE_DIR}/examples/autogenerated/zoneedit-activity/Main.idric")
set(dex "${stage}/classes.dex")
add_custom_command(OUTPUT "${dex}" "${CMAKE_BINARY_DIR}/zoneedit-classes.checked.anf"
  COMMAND "${CMAKE_COMMAND}" -E make_directory "${stage}" "${generator_dir}"
  COMMAND "${CMAKE_COMMAND}" -E rm -f "${dex}" "${generator_dir}/zoneedit-dex"
  COMMAND "${CMAKE_COMMAND}" -E env ${compiler_env} "${IDRIC_EXECUTABLE}"
    --source-dir "${DEX_BACKEND_DIR}/src" --build-dir "${CMAKE_BINARY_DIR}/backend"
    --check "${DEX_BACKEND_DIR}/src/Backend/DEX/FrameworkClasses.idr"
  COMMAND "${CMAKE_COMMAND}" -E env ${compiler_env} "${IDRIC_EXECUTABLE}"
    --source-dir "${CMAKE_SOURCE_DIR}/examples/autogenerated/zoneedit-activity"
    --build-dir "${CMAKE_BINARY_DIR}/checked" --cg idric-one-step
    --output-dir "${CMAKE_BINARY_DIR}" -o zoneedit-classes.checked.anf "${activity}"
  COMMAND "${CMAKE_COMMAND}" -E env ${compiler_env} "${IDRIC_EXECUTABLE}"
    --source-dir "${CMAKE_SOURCE_DIR}/examples/autogenerated/zoneedit-activity"
    --build-dir "${CMAKE_BINARY_DIR}/generator-build" --output-dir "${generator_dir}"
    -o zoneedit-dex "${activity}"
  COMMAND "${CMAKE_COMMAND}" -E env "LD_LIBRARY_PATH=${IDRIC_SUPPORT_DIR}/support/c"
    "${generator_dir}/zoneedit-dex" "${dex}"
  DEPENDS "${activity}" ${backend_sources}
  VERBATIM COMMENT "Check Idriç class descriptions and encode direct DEX")

set(libraries)
set(abis armeabi-v7a arm64-v8a x86_64)
set(triples armv7a-linux-androideabi26 aarch64-linux-android26 x86_64-linux-android26)
foreach(index RANGE 0 2)
  list(GET abis ${index} abi)
  list(GET triples ${index} triple)
  set(library "${stage}/lib/${abi}/libzoneedit.so")
  add_custom_command(OUTPUT "${library}"
    COMMAND "${CMAKE_COMMAND}" -E make_directory "${stage}/lib/${abi}"
    COMMAND "${ndk}/${triple}-clang" -std=c17 -O2 -Wall -Wextra -Werror -fPIC -shared
      -Wl,-z,defs -Wl,-z,max-page-size=16384
      "${CMAKE_SOURCE_DIR}/app/src/main/c/zoneedit.c"
      "${CMAKE_SOURCE_DIR}/app/src/main/c/navigation.c" -llog -o "${library}"
    DEPENDS app/src/main/c/zoneedit.c app/src/main/c/navigation.c
      app/src/main/c/navigation.h app/src/main/c/mobile-css.h
    VERBATIM COMMENT "NDK compile/link: ${abi}")
  list(APPEND libraries "${library}")
endforeach()

set(apk "${CMAKE_BINARY_DIR}/zoneedit-mobile.apk")
add_custom_command(OUTPUT "${apk}"
  COMMAND "${tools}/aapt2" link -I "${ANDROID_SDK_ROOT}/platforms/android-35/android.jar"
    --manifest "${CMAKE_SOURCE_DIR}/app/src/main/AndroidManifest.xml"
    --min-sdk-version 26 --target-sdk-version 35 --version-code 2 --version-name 0.2.0
    -o "${CMAKE_BINARY_DIR}/unsigned.apk"
  COMMAND touch -t 198001010000 "${dex}" ${libraries}
  COMMAND zip -X -0 -q "${CMAKE_BINARY_DIR}/unsigned.apk" classes.dex
    lib/armeabi-v7a/libzoneedit.so lib/arm64-v8a/libzoneedit.so lib/x86_64/libzoneedit.so
  COMMAND "${tools}/zipalign" -f -P 16 4 "${CMAKE_BINARY_DIR}/unsigned.apk" "${CMAKE_BINARY_DIR}/aligned.apk"
  COMMAND "${tools}/apksigner" sign --ks "${CMAKE_SOURCE_DIR}/_/signing/test-signer.p12"
    --ks-key-alias wegert-debug --ks-pass pass:wegert-debug --key-pass pass:wegert-debug
    --out "${apk}" "${CMAKE_BINARY_DIR}/aligned.apk"
  COMMAND "${tools}/apksigner" verify --verbose --print-certs "${apk}"
  COMMAND "${tools}/zipalign" -c -P 16 4 "${apk}"
  COMMAND "${CMAKE_COMMAND}" "-DAPK=${apk}" "-DTOOLS=${tools}" "-DNDK=${ndk}"
    "-DAICI_DIR=${AICI_DIR}" -P "${CMAKE_SOURCE_DIR}/_/verify-apk.cmake"
  COMMAND sha256sum "${apk}"
  WORKING_DIRECTORY "${stage}"
  DEPENDS "${dex}" ${libraries} app/src/main/AndroidManifest.xml _/signing/test-signer.p12
    _/verify-apk.cmake "${AICI_DIR}/android-signing/identities.tsv"
  VERBATIM COMMENT "Package, align, sign and verify ZoneEdit APK")
add_custom_target(zoneedit ALL DEPENDS "${apk}")

add_custom_target(navigation-test
  COMMAND "${ndk}/clang" --target=x86_64-linux-gnu --sysroot=/ -fuse-ld=lld
    -std=c17 -Wall -Wextra -Werror "${CMAKE_SOURCE_DIR}/tests/navigation.c"
    "${CMAKE_SOURCE_DIR}/app/src/main/c/navigation.c" -o "${CMAKE_BINARY_DIR}/navigation-test"
  COMMAND "${CMAKE_BINARY_DIR}/navigation-test"
  COMMAND "${CMAKE_COMMAND}" "-DSOURCE=${CMAKE_SOURCE_DIR}" "-DBUILD=${CMAKE_BINARY_DIR}"
    "-DNDK=${ndk}" -P "${CMAKE_SOURCE_DIR}/_/navigation-negative.cmake"
  VERBATIM COMMENT "Execute URL policy tests with the NDK host compiler")
